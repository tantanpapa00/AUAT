const string group_osc = "06. 지표 (원본 OSC)"
len_smooth = input.int(20, "스무딩 길이", minval=2, group=group_osc)
threshold  = input.float(1.0, "임계값(Threshold)", step=0.1, group=group_osc)
len_std    = 50

// smoother
smoother_F(src, len) =>
    float a = 2.0 / (len + 1)
    float b = 1 - a
    var float out = na
    out := na(out[1]) ? src : a * src + b * out[1]
    out

// std array (원본 구조 보존용)
float[] std_array = array.new<float>(len_std)

float line_long  = smoother_F(close, len_smooth * 2)
float line_short = smoother_F(close, len_smooth)
float oscillator = line_short - line_long

float stdev_osc  = ta.stdev(oscillator, len_std)
for int i = 0 to len_std - 1
    array.set(std_array, i, stdev_osc[i])

float denom = math.max(ta.highest(stdev_osc, len_std), 1e-10)
float normalized_osc = ta.hma(oscillator / denom, 30)

float basis      = ta.ema(normalized_osc, 250)
float deviation  = 2 * ta.stdev(normalized_osc, 250)
float upper_band = basis + deviation
float lower_band = basis - deviation

// Plot (원본 시각)
plot(upper_band, color=color.gray, editable=false)
plot(lower_band, color=color.gray, editable=false)

p1   = plot(threshold,  color=#787b867e, editable=false)
p_1  = plot(-threshold, color=#787b867e, editable=false)
p0   = plot(0,          color=color.gray, editable=false)
p4   = plot(4,  display=display.none, editable=false)
p_4  = plot(-4, display=display.none, editable=false)

plot_osc = plot(normalized_osc, color=chart.fg_color, editable=false)

fill(p0, p1,  threshold, 0, na, color.new(#278ed3, 70), editable=false)
fill(p0, p_1, 0, -threshold, color.new(#278ed3, 70), na, editable=false)
fill(p_1, p_4, -threshold, -4, na, color.new(color.aqua, 70), editable=false)
fill(p1, p4, 4, threshold, color.new(color.orange, 70), na, editable=false)
fill(p0, plot_osc, normalized_osc, 0, normalized_osc > 0 ? color.aqua : na, na, editable=false)
fill(p0, plot_osc, normalized_osc, 0, normalized_osc < 0 ? color.orange : na, na, editable=false)

// - 정책: '국면(연속 구간)당 1회'
// - fire: R3 상태에서 close가 HTF senkouB를 상향 돌파 (국면별 확정봉 규칙 적용)
// - reset: R3 이탈 시 used 리셋(다음 R3 구간에서 다시 1회 허용)
// - 옵션: 음운(SpanA<SpanB)일 때만 돌파 인정
// ------------------------------
bool r3_enter = (regime == 3) and (regime[1] != 3)
bool r3_exit  = (regime != 3) and (regime[1] == 3)
if r3_exit
    r3_break_used := false
if r3_enter
    r3_break_used := false

bool r3_cloud_ok = (not r3_breakout_require_cloud_bear) ? true : cloud_bear
bool r3_break_event = okbar_reg and r3_cloud_ok and (close > htf_senkouB) and (close[1] <= htf_senkouB[1])
bool breakout_trigger = (regime == 3) and r3_breakout_on and r3_break_event and (not r3_break_used)

// 국면별 OSC 매수 허용
bool reg_allow_osc_buy = switch regime
    1 => r1_allow_osc_buy
    2 => r2_allow_osc_buy
    3 => r3_allow_osc_buy
    4 => r4_allow_osc_buy
    => false

bool any_buy_trigger = (osc_trigger and reg_allow_osc_buy) or pullback_trigger or breakout_trigger

bool can_buy = allowTrade and buy_lock_ok and allow_buy_by_stage and gap_ok and below_avg_ok and prev_sig_ok and prev_exec_ok and any_buy_trigger and (reg_buy_mult > 0)

if can_buy
    float tranche_pct = 0.0
    bool is_breakout_buy = breakout_trigger
    bool is_pullback_buy = pullback_trigger

    if is_breakout_buy
        tranche_pct := buy1 * r3_breakout_buy_mult * reg_buy_mult
    else if is_pullback_buy
        tranche_pct := buy1 * r1_pullback_buy_mult * reg_buy_mult
    else
        tranche_pct := f_getBuyPct(eff_stage) * reg_buy_mult

    float available_cash = math.max(strategy.equity - pos_value, 0)
    float spend_base = available_cash * (cash_use_pct / 100.0)
    float spend = spend_base * (tranche_pct / 100.0)

    bool ok_cap = (pos_value + spend) <= max_pos_val

    if tranche_pct > 0 and spend > 0 and ok_cap
        float buy_qty = spend / close
        if buy_qty > 0
            // Webhook payload
            string amsg = f_buyMsg(INST_ID, buy_qty)

            strategy.entry(ENTRY_ID, strategy.long, qty=buy_qty, alert_message=(use_tv_alert_msg ? amsg : ""))

            last_buy_time := time
            last_buy_exec_price := close
            last_buy_signal_price := close

            if pullback_trigger
                r1_pb_used := true
                r1_pb_armed := false
            if breakout_trigger
                r3_break_used := true

            if inv_active
                float oldq = nz(inv_qty, 0.0)
                float newq = oldq + buy_qty
                inv_avg := newq > 0 ? ((na(inv_avg) or oldq == 0) ? close : (inv_avg * oldq + close * buy_qty) / newq) : na
                inv_qty := newq

            if reg_buy1_only
                buy_stage := 0
            else
                if after_max_buy == "cycle"
                    buy_stage := (buy_stage >= lastStage) ? 0 : (buy_stage + 1)
                else if after_max_buy == "stop"
                    buy_stage := (buy_stage >= lastStage) ? lastStage : (buy_stage + 1)
                else
                    buy_stage := (buy_stage >= lastStage) ? lastStage : (buy_stage + 1)

            if reg_reset_sell_on_buy
                sell_stage := 0
    else
        if show_debug and debug_verbose and (osc_trigger or breakout_trigger)
            string _rsn = tranche_pct <= 0 ? "매수비중 0%" : spend <= 0 ? "가용자금 부족" : not ok_cap ? "절대 노출 상한" : "조건 불충족"
            f_dbgFail(regime, _rsn, normalized_osc, true)

if allowTrade and signal_up
    last_buy_signal_price := close

//==================================================
// 16) SELL 로직 (교대매도 포함)
//==================================================
float need_pct = min_profit_pct + fee_roundtrip_buf
bool sell_ok_price = strat_has_pos and close >= avg_price * (1 + need_pct / 100.0)

int lastSellStage = math.max(max_sell_tranches - 1, 0)
int eff_sell_stage = reg_sell1_only ? 0 : sell_stage

if not reg_sell1_only
    if eff_sell_stage >= max_sell_tranches
        if after_max_sell == "cycle"
            eff_sell_stage := 0
        else
            eff_sell_stage := lastSellStage

bool allow_sell_by_stage = true
if not reg_sell1_only and after_max_sell == "stop" and sell_stage >= max_sell_tranches
    allow_sell_by_stage := false

bool alt_allow = true
if reg_sell_mode == "Alternate" and signal_dn and strat_has_pos
    alt_sell_toggle := not alt_sell_toggle
    alt_allow := alt_sell_toggle

bool can_sell = allowTrade and sell_lock_ok and signal_dn and strat_has_pos and sell_ok_price and allow_sell_by_stage and alt_allow and (reg_sell_mult > 0)

if can_sell
    float spct = f_getSellPctSingle(eff_sell_stage) * reg_sell_mult
    spct := math.min(spct, 100.0)

    if spct > 0
        float qty_target = strategy.position_size * (spct / 100.0)
        qty_target := math.min(qty_target, strategy.position_size)

        if qty_target > 0
            string smsg = f_sellMsg(INST_ID, qty_target)
            strategy.close(ENTRY_ID, qty=qty_target, alert_message=(use_tv_alert_msg ? smsg : ""))

            last_sell_time := time

            if inv_active
                float newq2 = math.max(nz(inv_qty, 0.0) - qty_target, 0.0)
                inv_qty := newq2
                if newq2 == 0
                    inv_avg := na

            if reg_sell1_only
                sell_stage := 0
            else
                if after_max_sell == "cycle"
                    sell_stage := (sell_stage >= lastSellStage) ? 0 : (sell_stage + 1)
                else if after_max_sell == "stop"
                    sell_stage := (sell_stage >= lastSellStage) ? lastSellStage : (sell_stage + 1)
                else
                    sell_stage := (sell_stage >= lastSellStage) ? lastSellStage : (sell_stage + 1)
    else
        if show_debug
            f_dbgFail(regime, "매도비중 0%", normalized_osc, false)
else
    if show_debug and signal_dn and strat_has_pos and not sell_ok_price
        f_dbgFail(regime, "최소익절 게이트 미충족", normalized_osc, false)

//==================================================
// 16.5) 표시 동기화: 신호(조건) vs 실제 체결(포지션 변화)
// - 차트 마커/백테스트 불일치는 대부분 '신호는 떴지만 주문은 안나감(게이트/락/익절조건)' 또는
//   '실시간봉에서 신호가 깜빡(확정 전)했다가 사라짐' 때문입니다.

//==================================================

bool didBuy  = (strategy.position_size > strategy.position_size[1])
bool didSell2 = (strategy.position_size < strategy.position_size[1])

plotshape(show_exec_marks and didBuy,  title='EXEC BUY',  style=shape.labelup,   text='BUY',  size=size.tiny, location=location.belowbar, color=color.new(color.aqua,0), textcolor=color.white, force_overlay=true)
plotshape(show_exec_marks and didSell2, title='EXEC SELL', style=shape.labeldown, text='SELL', size=size.tiny, location=location.abovebar, color=color.new(color.orange,0), textcolor=color.white, force_overlay=true)

//==================================================
// 17) SELL 실행 감지 -> BUY 사이클 리셋 (국면별)
//==================================================
bool didSell = (strategy.position_size < strategy.position_size[1])
if didSell and reg_reset_buy_on_sell
    buy_stage := 0
    last_buy_signal_price := na

f_join(string[] _arr, string _sep) =>
    string _out = ""
    int _n = array.size(_arr)
    for _i = 0 to _n - 1
        _out := _out + (_i == 0 ? "" : _sep) + array.get(_arr, _i)
    _out

if show_debug and debug_verbose and signal_up and not can_buy
    string[] _rs = array.new_string()
    if not buy_lock_ok
        array.push(_rs, "봉당 1회 제한")
    if not allow_buy_by_stage
        array.push(_rs, "BUY 스테이지 제한")
    if not lower_band_ok and osc_trigger
        array.push(_rs, "하단밴드 조건X")
    if has_pos and use_add_buy_gap and not gap_ok

        array.push(_rs, "직전 신호가 하락X")
    if has_pos and reg_filt_prev_exec and not prev_exec_ok
        array.push(_rs, "직전체결가 하락X")
    if (not osc_trigger) and (not breakout_trigger) and (not pullback_trigger)
        array.push(_rs, "신호 없음")
    if reg_buy_mult <= 0
        array.push(_rs, "국면 매수비중 0")

    if (regime == 1) and r1_pullback_on and (not r1_pb_used)
        if not r1_pb_armed and htf_hull_dn
            array.push(_rs, "R1 눌림: armed 대기")

    if array.size(_rs) == 0

if show_debug and debug_verbose and signal_dn and not can_sell
    string[] _rs2 = array.new_string()
    if not sell_lock_ok
        array.push(_rs2, "봉당 1회 제한")

        array.push(_rs2, "최소익절 게이트X")
    if reg_sell_mode == "Alternate" and not alt_allow and strat_has_pos
        array.push(_rs2, "교대매도: 이번 신호 스킵")
    if reg_sell_mult <= 0
        array.push(_rs2, "국면 매도비중 0")
    if array.size(_rs2) == 0

if barstate.islast
    float avg  = strategy.position_avg_price
    float qty  = math.abs(strategy.position_size)
    float pnl  = qty > 0 ? (close - avg) * qty : 0.0
    float pnlP = qty > 0 ? (close / avg - 1.0) * 100.0 : 0.0

    if showHud
        if hudMode == "TABLE(패널)"
            table.clear(hudT, 0, 0, 1, 3)

            table.cell(hudT, 0, 0, "항목", text_size=size.tiny)
            table.cell(hudT, 1, 0, "값",  text_size=size.tiny)

            table.cell(hudT, 0, 1, "평단가", text_size=size.tiny)
            table.cell(hudT, 1, 1, qty > 0 ? f_fmt(avg) : "-", text_size=size.tiny)

            table.cell(hudT, 0, 2, "보유수량", text_size=size.tiny)
            table.cell(hudT, 1, 2, qty > 0 ? str.tostring(qty) : "0", text_size=size.tiny)

            table.cell(hudT, 0, 3, "미실현손익", text_size=size.tiny)
            table.cell(hudT, 1, 3, qty > 0 ? (str.tostring(pnl, format.mintick) + " (" + str.tostring(pnlP, format.mintick) + "%)") : "-", text_size=size.tiny)
        else
            if not na(hudL)
                label.delete(hudL)

            float y = high + ta.atr(atrLenHud) * 2.0
            string txt =
                 "평단가: " + (qty > 0 ? f_fmt(avg) : "-") + "\n" +
                 "보유수량: " + (qty > 0 ? str.tostring(qty) : "0") + "\n" +
                 "미실현: " + (qty > 0 ? (str.tostring(pnl, format.mintick) + " (" + str.tostring(pnlP, format.mintick) + "%)") : "-")

            hudL := label.new(bar_index, y, txt,
                 xloc=xloc.bar_index, yloc=yloc.price,
                 style=label.style_label_upper_right, size=size.small)
    else
        table.clear(hudT, 0, 0, 1, 3)
        if not na(hudL)
            label.delete(hudL)
