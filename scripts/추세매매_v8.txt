// Pyramid weight helper
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
f_getPyrWeight(int stage) =>
    switch stage
        0 => pyrWt1
        1 => pyrWt2
        2 => pyrWt3
        3 => pyrWt4
        4 => pyrWt5
        5 => pyrWt6
        6 => pyrWt7
        7 => pyrWt8
        8 => pyrWt9
        9 => pyrWt10
        => 0.0

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// Position sizing (피라미딩 차수별 수량 계산)
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
f_calcQty(float weightPct) =>
    float cash = strategy.equity * (posPct / 100.0) * (weightPct / 100.0)
    float raw = cash / close
    float q = roundQty ? math.floor(raw) : raw
    math.max(minQty, q)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// Supertrend
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// HTF Filter (주식: 주봉 VWMA156 / 코인: 일봉 MA200 / 수동 설정)
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

     useLastConfirmedHTF ? htfExpr[1] : htfExpr,
     barmerge.gaps_off, barmerge.lookahead_off)

bool htfOk = true

if useHTFFilter and useWeeklyCloseRegime
    bool newPeriod = timeframe.change(htfTF)
    float prevClose = request.security(syminfo.tickerid, htfTF, close[1], barmerge.gaps_off, barmerge.lookahead_off)
    float prevHTF   = request.security(syminfo.tickerid, htfTF, htfExpr[1], barmerge.gaps_off, barmerge.lookahead_off)
    if newPeriod or barstate.isfirst
        wRegimeOK := (not na(prevClose) and not na(prevHTF)) ? (prevClose > prevHTF) : false
    htfOk := wRegimeOK
else
    htfOk := not useHTFFilter ? true : (close > htfValue and not na(htfValue))

plot(useHTFFilter ? htfValue : na, "HTF MA", color=color.new(color.orange, 0), linewidth=2)

// HVI (LazyBear)
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
rng    = high - low
rngAvg = ta.sma(rng, hviLen)
volA   = ta.sma(volume, hviLen)

high1 = high[1]
low1  = low[1]
mid1  = hl2[1]

u1 = mid1 + (high1 - low1) / hviDiv
d1 = mid1 - (high1 - low1) / hviDiv

r_enabled1 = (rng > rngAvg) and (close < d1) and (volume > volA)
r_enabled2 = close < mid1
r_enabled  = r_enabled1 or r_enabled2

g_enabled1 = close > mid1
g_enabled2 = (rng > rngAvg) and (close > u1) and (volume > volA)
g_enabled3 = (high > high1) and (rng < rngAvg / 1.5) and (volume < volA)
g_enabled4 = (low < low1)  and (rng < rngAvg / 1.5) and (volume > volA)
g_enabled  = g_enabled1 or g_enabled2 or g_enabled3 or g_enabled4

gr_enabled1 = (rng > rngAvg) and (close > d1) and (close < u1) and (volume > volA) and (volume < volA * 1.5) and (volume > volume[1])
gr_enabled2 = (rng < rngAvg / 1.5) and (volume < volA / 1.5)
gr_enabled3 = (close > d1) and (close < u1)
gr_enabled  = gr_enabled1 or gr_enabled2 or gr_enabled3

hviColor = gr_enabled ? color.gray : g_enabled ? color.green : r_enabled ? color.red : color.blue
plot(showHVIPane ? volume : na, "HVI Volume", style=plot.style_columns, color=hviColor, linewidth=2, display=display.pane)

hviGreen = g_enabled

// QQE MOD
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
calculateQQE(rsiLength, smoothingFactor, qqeFactor, source) =>
    wildersLength = rsiLength * 2 - 1
    rsi = ta.rsi(source, rsiLength)
    smoothedRsi = ta.ema(rsi, smoothingFactor)
    atrRsi = math.abs(smoothedRsi[1] - smoothedRsi)
    smoothedAtrRsi = ta.ema(atrRsi, wildersLength)
    dynamicAtrRsi = smoothedAtrRsi * qqeFactor

    longBand = 0.0
    shortBand = 0.0
    trendDirection = 0

    atrDelta = dynamicAtrRsi
    newShortBand = smoothedRsi + atrDelta
    newLongBand  = smoothedRsi - atrDelta

    longBand := smoothedRsi[1] > longBand[1] and smoothedRsi > longBand[1] ? math.max(longBand[1], newLongBand) : newLongBand
    shortBand := smoothedRsi[1] < shortBand[1] and smoothedRsi < shortBand[1] ? math.min(shortBand[1], newShortBand) : newShortBand

    longBandCross = ta.cross(longBand[1], smoothedRsi)

    if ta.cross(smoothedRsi, shortBand[1])
        trendDirection := 1
    else if longBandCross
        trendDirection := -1
    else
        trendDirection := trendDirection[1]

    qqeTrendLine = trendDirection == 1 ? longBand : shortBand
    [qqeTrendLine, smoothedRsi]

[primaryQQETrendLine, primaryRSI]     = calculateQQE(rsiLengthPrimary, rsiSmoothingPrimary, qqeFactorPrimary, sourcePrimary)
[secondaryQQETrendLine, secondaryRSI] = calculateQQE(rsiLengthSecondary, rsiSmoothingSecondary, qqeFactorSecondary, sourceSecondary)

bollBasis = ta.sma(primaryQQETrendLine - 50, bollingerLength)
bollDev   = bollingerMultiplier * ta.stdev(primaryQQETrendLine - 50, bollingerLength)
bollUpper = bollBasis + bollDev

qqeUpState = (secondaryRSI - 50 > thresholdSecondary) and (primaryRSI - 50 > bollUpper)

qqePos = switch qqePositiveDef
    "PrimaryRSI>50"        => primaryRSI > 50
    "SecondaryRSI>50"      => secondaryRSI > 50
    "Secondary>0(hist)"    => (secondaryRSI - 50) > 0
    "QQE UpState(blue)"    => qqeUpState

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// SPO (Smooth Price Oscillator)
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
smoother_F(float price, int period) =>
    float step = 2.0 * math.pi / period
    float a1 = math.exp(-math.sqrt(2) * math.pi / period)
    float b1 = 2 * a1 * math.cos(math.sqrt(2) * step / period)
    float c2 = b1
    float c3 = -a1 * a1
    float c1 = 1 - c2 - c3
    float smoothed = 0.0
    smoothed := bar_index >= 4 ? c1 * (price + price[1]) / 2 + c2 * smoothed[1] + c3 * smoothed[2] : price
    smoothed

float line_long  = smoother_F(close, spo_len_smooth * 2)
float line_short = smoother_F(close, spo_len_smooth)
float oscillator = line_short - line_long

float stdev_osc  = ta.stdev(oscillator, spo_len_std)
float denom      = math.max(ta.highest(stdev_osc, spo_len_std), 1e-10)
float norm_osc   = ta.hma(oscillator / denom, spo_hma_norm)

bool spo_okbar = spo_confirmed ? barstate.isconfirmed : true
bool spo_signal_dn = spo_okbar and (norm_osc > spo_threshold) and ta.crossover(norm_osc[1], norm_osc)

plot(showSpoPlot ? norm_osc : na, "SPO normalized", display=display.pane)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// ENTRY CONDITIONS
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
buySignal = stBull and hviGreen and qqePos and htfOk

setupState = buySignal
setupStart = setupState and not setupState[1]
setupEnd   = not setupState and setupState[1]

entrySignal = enterOnlyOnSetupStart ? setupStart : buySignal

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// LIVE GUARD
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
var bool liveArmed   = false
var bool lockCurrent = false

justArmed = useLiveGuard and (not liveArmed) and (time >= tradeStart)
if justArmed
    liveArmed   := true
    lockCurrent := setupState

if liveArmed and setupEnd
    lockCurrent := false

entryWindowOK = not useLiveGuard ? true : (liveArmed and not lockCurrent)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// PYRAMIDING STATE
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
var int pyrCount = 0
var int lastPyrBar = na
var float pyrHighest = na

// 포지션 없으면 리셋
if strategy.position_size == 0
    pyrCount := 0
    lastPyrBar := na
    pyrHighest := na

// 포지션 있을 때 최고가 추적
if strategy.position_size > 0
    pyrHighest := na(pyrHighest) ? high : math.max(pyrHighest, high)

// N봉 신고가 (lookback 기준)
highestN = ta.highest(high[1], pyrHighLen)

// 피라미딩 조건: 현재가가 N봉 신고가를 돌파 + 쿨다운 + 4조건 유지 + 최대 횟수 미만
bool pyrCooldownOK = na(lastPyrBar) or (bar_index - lastPyrBar >= pyrCooldown)
bool pyrBreakout   = close > highestN and not na(highestN)
bool canPyramid    = usePyramiding and strategy.position_size > 0 and pyrCount < maxPyrEntries and pyrBreakout and pyrCooldownOK and stBull and htfOk

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// SELL STATE
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
f_getSellPctSingle(stage) =>
    switch stage
        0 => sell1
        1 => sell2
        2 => sell3
        3 => sell4
        4 => sell5
        5 => sell6
        => sell6

var int sell_stage = 0
var int last_sell_bar = na
var bool tp1_done = false

if strategy.position_size == 0
    sell_stage := 0
    last_sell_bar := na
    tp1_done := false

need_pct = min_profit_pct + fee_buf_pct
sell_ok_price = not use_profit_gate ? true : (strategy.position_size > 0 and close >= strategy.position_avg_price * (1 + need_pct / 100.0))

sell_lock_ok = not one_sell_per_bar ? true : (na(last_sell_bar) or last_sell_bar != bar_index)

can_split_sell = useSPO and strategy.position_size > 0 and spo_signal_dn and sell_ok_price and sell_lock_ok

// TP1
tp1PriceHit = strategy.position_size > 0 and useTP1 and not tp1_done and close >= strategy.position_avg_price * (1 + tp1ProfitPct / 100.0)
can_tp1_sell = tp1PriceHit and sell_lock_ok

// Supertrend 전량매도 (현재TF or 상위TF 선택)
exitAllSignal = switch stExitMode
    "Current TF"      => stBearFlip
    "HTF only"        => stHTF_FlipDown
    "Both (any flip)" => stBearFlip or stHTF_FlipDown
    "None"            => false

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// ATR for dynamic stop
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
atrVal = ta.atr(atrStopLen)

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// ORDERS
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// ✅ 1차 진입
realEntryCond = entryWindowOK and entrySignal and strategy.position_size == 0 and barstate.isconfirmed

if realEntryCond
    float qty1 = f_calcQty(f_getPyrWeight(0))
    if qty1 > 0
        strategy.entry("Long", strategy.long, qty=qty1, comment="ENTRY 1")
        pyrCount := 1
        lastPyrBar := bar_index
        pyrHighest := high

// ✅ 피라미딩 (2차~ 추가매수)
if canPyramid and barstate.isconfirmed
    float pyrQty = f_calcQty(f_getPyrWeight(pyrCount))
    if pyrQty > 0
        strategy.entry("Long", strategy.long, qty=pyrQty, comment="PYR " + str.tostring(pyrCount + 1))
        pyrCount += 1
        lastPyrBar := bar_index

// ✅ Stop Loss (고정% / ATR 기반)
if useStopLoss and strategy.position_size > 0
    float stopPrice = na
    if stopType == "Fixed %"
        stopPrice := strategy.position_avg_price * (1 - stopLossPct / 100.0)
    else
        stopPrice := strategy.position_avg_price - (atrVal * atrStopMult)

    if not na(stopPrice) and stopPrice > 0
        strategy.exit("SL", from_entry="Long", stop=stopPrice, comment=stopType == "Fixed %" ? "HARD SL" : "ATR SL")

// ✅ TP1
if can_tp1_sell
    float tpQty = strategy.position_size * (tp1SellPct / 100.0)
    tpQty := math.min(tpQty, strategy.position_size)
    if tpQty > 0
        strategy.close("Long", qty=tpQty, comment="TP1")
        tp1_done := true
        last_sell_bar := bar_index
        if pyrRefillAfterSell
            pyrCount := math.max(pyrCount - 1, 1)

// ✅ 분할매도 (SPO signal_dn)
if can_split_sell and not can_tp1_sell
    int lastStage = math.max(max_sell_tranches - 1, 0)
    int eff_stage = sell_stage

    if eff_stage >= max_sell_tranches
        if after_max_sell == "cycle"
            eff_stage := 0
        else
            eff_stage := lastStage

    bool allow_sell_by_stage = true
    if after_max_sell == "stop" and sell_stage >= max_sell_tranches
        allow_sell_by_stage := false

    if allow_sell_by_stage
        float spct = f_getSellPctSingle(eff_stage)
        if spct > 0
            float qty_target = strategy.position_size * (spct / 100.0)
            qty_target := math.min(qty_target, strategy.position_size)

            if qty_target > 0
                strategy.close("Long", qty=qty_target, comment="SPO S" + str.tostring(eff_stage + 1))
                last_sell_bar := bar_index
                if pyrRefillAfterSell
                    pyrCount := math.max(pyrCount - 1, 1)

                if after_max_sell == "cycle"
                    sell_stage := sell_stage >= lastStage ? 0 : sell_stage + 1
                else
                    sell_stage := sell_stage >= lastStage ? lastStage : sell_stage + 1

// ✅ 전량매도: Supertrend 하락 전환
if strategy.position_size > 0 and exitAllSignal
    strategy.close("Long", comment="EXIT ALL: ST Flip")
